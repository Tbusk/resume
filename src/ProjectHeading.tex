\newcommand{\rProjectHeading}[4]{
    \begin{tabular*}{
        \textwidth % whole doc width
        \vspace{4pt}
    }{
        l@{\extracolsep{\fill}}r % separates cols with spacing
    }
    \if\relax#2\relax
        \if\relax#3\relax
            \hspace{4pt}\textbf{#1} & {\small#4}\\
        \else
            \hspace{4pt}\textbf{#1} \small{(\textit{\href{https://#3}{#3}})} & {\small#4} \\
        \fi
    \else
        \if\relax#3\relax
            \hspace{4pt}\textbf{#1,} {#2} & {\small#4}\\
        \else
            \hspace{4pt}\textbf{#1,} {#2} \small{(\textit{\href{https://#3}{#3}})} & {\small#4} \\
        \fi
    \fi
    \end{tabular*}\vspace{-12pt}
}
